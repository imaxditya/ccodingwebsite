Question 2

Write a program in which we are having 2D array of 2 by 3.
Insert the value through pointer & print the values

Coding

#include stdio.h
    #include dos.h 
    #include conio.h    
int arr[2][3], *ptr, i,j;
clsrcr();
ptr=arr[0];
for(i=0; i<=1; i++)
{
	for(j=0; j<=2; j++)
	{
	printf("Enter the value:");
	scanf("%d",&arr[i][j]);
	}
}
for(i=0; i<=1; i++)
{
	for(j=0; j<=2; j++)
	{
	printf("%u",*ptr);
	ptr++;
	}
printf("\n");
}
delay(3000);

Output

Enter the value:1,2,3,4,5,6
1  2  3
4  5  6
