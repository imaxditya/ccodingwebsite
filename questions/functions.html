Question 4

Write a program to swap two numbers using call by address.

Coding

#include stdio.h

void swap(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}

void main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    swap(&a, &b);

    printf("After swapping:\n");
    printf("a = %d\n", a);
    printf("b = %d", b);
}

Output

Enter two numbers: 10 20
After swapping:
a = 20
b = 10

Question 6

Write a C program to make a function to find the power of a number. Enter the number and power from the user.

Coding

#include stdio.h

int power(int num, int p)
{
    int result = 1;

    for(int i = 1; i <= p; i++)
    {
        result = result * num;
    }

    return result;
}

int main()
{
    int num, p, result;

    printf("Enter a number: ");
    scanf("%d", &num);

    printf("Enter power: ");
    scanf("%d", &p);

    result = power(num, p);

    printf("Answer = %d", result);

    return 0;
}

Output

Enter a number: 2
Enter power: 3
Answer = 8
