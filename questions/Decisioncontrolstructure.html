Question 18

Write a program to enter date, month, and year from the user and check whether the date is valid or not.

Coding

#include stdio.h

void main()
{
    int d, m, y, days;

    printf("Enter date, month and year: ");
    scanf("%d %d %d", &d, &m, &y);

    if (y < 1 || m < 1 || m > 12)
    {
        printf("Invalid Date");
    }
    else
    {
        if (m == 2)
        {
            if (y % 400 == 0 || (y % 4 == 0 && y % 100 != 0))
                days = 29;
            else
                days = 28;
        }
        else if (m == 4 || m == 6 || m == 9 || m == 11)
        {
            days = 30;
        }
        else
        {
            days = 31;
        }

        if (d >= 1 && d <= days)
            printf("Valid Date");
        else
            printf("Invalid Date");
    }
}

Output

Enter date, month and year: 29 2 2024
Valid Date
