Question 2

Enter the filename from the user and enter the data from the user and put in the file but data 
should be in capital letter.

Coding

#include conio.h
#include dos.h

int i;

printf("Enter filename:");
scanf("%s",fname);

f=fopen("fname","w");

Question 4

Enter the filename from the user and enter the data from the user and copy that file into another file

Coding

#include conio.h
#include dos.h

char fname1[20], fname2[20], data[100];
int ch;
clrscr();
printf("Enter filename:");
scanf("%s",fname1);

f1=fopen("fname","w");
printf("Enter data:");
scanf("%s",data);

fprintf(f1,"%s",data);
fclose(f1);

printf("Enter second filename:");
scanf("%s",fname2);

f1=fopen("fname1","r");
f2=fopen("fname2","w");

while((ch=fgetc(f1))!=EOF)
{
fputc(ch,f2)
}
printf("File copied successfully");
fclose(f1);
fclose(f2);
delay(3000);
}

Output

Enter filename:abc;
Enter data:hello
Enter second filename: xyz
File copied successfully
