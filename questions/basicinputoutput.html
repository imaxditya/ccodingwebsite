Question 10

Enter two character from user and print ASCII value?

Coding

#include stdio.h
#include conio.h

void main()
{
    char ch1, ch2;

    clrscr();

    printf("Enter two characters: ");
    scanf("%c %c", &ch1, &ch2);

    printf("ASCII value of %c = %d\n", ch1, ch1);
    printf("ASCII value of %c = %d", ch2, ch2);

    getch();
}
