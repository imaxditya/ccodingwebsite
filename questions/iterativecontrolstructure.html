Question 1

Enter a number from the user and check it is a prime number or not.

Coding

#include stdio.h

void main()
{
    int n, i, count = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        if(n % i == 0)
        {
            count++;
        }
    }

    if(count == 2)
    {
        printf("Prime Number");
    }
    else
    {
        printf("Not a Prime Number");
    }
}

Output

Enter a number: 7
Prime Number

Question 21

    
Enter a number from the user and check whether it is an Armstrong number or not.


Coding



#include stdio.h
#include conio.h 
#include dos.h

void main()
{
    int n, temp, r, sum=0;
    clrscr();

    printf("Enter a number: ");
    scanf("%d",&n);

    temp=n;

    while(n>0)
    {
        r=n%10;
        sum=sum+(r*r*r);
        n=n/10;
    }

    if(temp==sum)
        printf("Armstrong number");
    else
        printf("Not an Armstrong number");

    delay(5000);
}



 
Output



Enter a number: 153
Armstrong Number
